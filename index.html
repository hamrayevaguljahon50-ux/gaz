import csv
import json
import smtplib
from email.mime.text import MIMEText
from datetime import datetime

# Sample Tank Storage Data structure
INITIAL_TANKS = {
    "GASOLINE_92": {"name": "Gasoline A-92", "capacity": 40000, "stock": 24500, "min": 8000, "unit": "L"},
    "GASOLINE_95": {"name": "Gasoline A-95", "capacity": 30000, "stock": 18200, "min": 6000, "unit": "L"},
    "DIESEL":      {"name": "Diesel Fuel",   "capacity": 50000, "stock": 38900, "min": 10000, "unit": "L"},
    "PROPANE_LPG": {"name": "Propane LPG",  "capacity": 25000, "stock": 4100,  "min": 5000, "unit": "Kg"},
    "CNG_METHANE": {"name": "CNG Methane",  "capacity": 15000, "stock": 9800,  "min": 3000, "unit": "m3"}
}

def calculate_current_stock(tanks, inflows, outflows):
    """ Formula: Current Stock = Initial Calibration + Inflows - Outflows """
    updated = tanks.copy()
    
    for item in inflows:
        fuel = item.get("fuel")
        if fuel in updated:
            updated[fuel]["stock"] += float(item.get("qty", 0))
            
    for item in outflows:
        fuel = item.get("fuel")
        if fuel in updated:
            updated[fuel]["stock"] -= float(item.get("qty", 0))
            
    return updated

def generate_shift_report(tanks):
    """ Format shift report text for email body or printing """
    now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    report = f"--- REFUELING STATION SHIFT REPORT [{now}] ---\n\n"
    
    for key, data in tanks.items():
        pct = (data["stock"] / data["capacity"]) * 100
        status = "CRITICAL LOW" if data["stock"] <= data["min"] else "OK"
        report += f"Fuel: {data['name']}\n"
        report += f"  Stock: {data['stock']:,} {data['unit']} / {data['capacity']:,} {data['unit']} ({pct:.1f}%)\n"
        report += f"  Status: [{status}]\n\n"
        
    return report

def send_smtp_alert(recipient_email, subject, body, smtp_host="smtp.gmail.com", smtp_port=587, user="", password=""):
    """ Dispatch automated alert via standard Python SMTP client """
    try:
        msg = MIMEText(body)
        msg['Subject'] = subject
        msg['From'] = user
        msg['To'] = recipient_email

        with smtplib.SMTP(smtp_host, smtp_port) as server:
            server.starttls()
            server.login(user, password)
            server.send_message(msg)
        print("[+] Email alert dispatched successfully.")
    except Exception as e:
        print(f"[-] Failed to dispatch email alert: {e}")

if __name__ == "__main__":
    print("[*] Processing Refueling Station Data Offline...")
    report = generate_shift_report(INITIAL_TANKS)
    print(report)
